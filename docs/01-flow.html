Follow a question

One question takes two paths through this repo: an offline build that turns a PDF into searchable chunks, and an online path that turns the question into a cited answer. This page follows the online path, stage by stage, using the real functions.

Worked example — one question, seven steps

Question: “Where was the bodhisatta born?”

This is the real trace from demo 01, not a retelling. Each step below is expandable; the long ones start collapsed. Open step 7 to see the exact excerpt block that becomes the prompt.

01_trace_question

Source: 01_trace_question.py (expand) 
"""Trace one question through retrieval without calling an LLM.

Step 1: read saved Settings, then find the dense index actually on disk.
Step 2: load the chunks that every score refers back to.
Step 3: score the question with TF-IDF (sparse, lexical).
Step 4: score it with fastembed (dense, semantic) when a dense index exists.
Step 5: min-max both score lists, then fuse with hybrid_weight.
Step 6: rank top-k; print lexical-only beside the mode the generator gets.
Step 7: print the exact numbered excerpt block that becomes the user message.

Run from the repo root:
    .venv/bin/python demos/01_trace_question.py
    .venv/bin/python demos/01_trace_question.py "What did Asita see?"

Scores are cosine similarities rounded to 2 decimals for readability.
Fused scores are relative ranks, not probabilities. No chat model is called.
"""

import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

import numpy as np

from rag.config import Settings
from rag.generate import SYSTEM_WITH_CONTEXT, format_context
from rag.index import DenseIndex, LexicalIndex
from rag.retrieve import Retriever, _minmax

QUESTION = sys.argv[1] if len(sys.argv) > 1 else "Where was the bodhisatta born?"

settings = Settings.load()
print("== step 1: configuration ==")
print(
    f"  saved defaults: llm={settings.llm_backend}/{settings.llm_model} | "
    f"embed={settings.embed_backend} | top_k={settings.top_k} | "
    f"hybrid_weight={settings.hybrid_weight:.2f}"
)

dense_files = sorted(settings.index_dir.glob("dense-*.npy"))
has_dense = bool(dense_files)
if has_dense:
    backend = dense_files[0].stem.removeprefix("dense-").split("-")[0]
    print(f"  dense file on disk: {dense_files[0].name} -> backend={backend}")
    settings = settings.overrides(embed_backend=backend)
    print(f"  demo uses embed={settings.embed_backend} so dense_path() resolves to that file")
else:
    print("  no dense-*.npy found: this demo will run lexical only")

retriever = Retriever(settings)
chunks = retriever.chunks
print("\n== step 2: chunks ==")
print(f"  loaded {len(chunks)} chunks from {settings.chunks_path.name}")
print(f"  example: {chunks[0].id} [{chunks[0].citation}]")

lexical = LexicalIndex(settings).load()
lex_scores = lexical.scores(QUESTION)
best_lex = int(lex_scores.argmax())
print("\n== step 3: lexical scores (TF-IDF cosine) ==")
print(
    f"  vector shape: {lex_scores.shape} | min={lex_scores.min():.2f} | "
    f"max={lex_scores.max():.2f} | nonzero={int((lex_scores > 0).sum())}"
)
print(f"  best chunk: {chunks[best_lex].id} [{chunks[best_lex].citation}]")

if has_dense:
    dense = DenseIndex(settings).load()
    dense_scores = dense.scores(QUESTION)
    best_dense = int(dense_scores.argmax())
    print("\n== step 4: dense scores (embedding cosine) ==")
    print(
        f"  vector shape: {dense_scores.shape} | min={dense_scores.min():.2f} | "
        f"max={dense_scores.max():.2f}"
    )
    print(f"  best chunk: {chunks[best_dense].id} [{chunks[best_dense].citation}]")
else:
    dense_scores = None
    print("\n== step 4: dense scores ==")
    print("  skipped: no dense index file for this embed backend")

k = settings.top_k
if dense_scores is None:
    final = lex_scores
    print("\n== step 5: no fusion; lexical scores are the final scores ==")
else:
    lex_norm = _minmax(lex_scores)
    dense_norm = _minmax(dense_scores)
    weight = settings.hybrid_weight
    final = (1 - weight) * lex_norm + weight * dense_norm
    print("\n== step 5: min-max fusion ==")
    print(
        f"  lex_norm: {lex_norm.min():.2f}..{lex_norm.max():.2f} | "
        f"dense_norm: {dense_norm.min():.2f}..{dense_norm.max():.2f}"
    )
    print(f"  final = {1 - weight:.2f} * lex_norm + {weight:.2f} * dense_norm")
print(f"  fused top-{k} indices (manual argsort): {np.argsort(-final)[:k].tolist()}")

lexical_results = retriever.search(QUESTION, k=k, mode="lexical")
mode = "hybrid" if has_dense else "lexical"
results = retriever.search(QUESTION, k=k, mode=mode)
print(f"\n== step 6: top-{k} ranking for \"{QUESTION}\" ==")
print("  lexical only:")
for rank, item in enumerate(lexical_results, start=1):
    print(f"    {rank}. {item.chunk.id} score={item.score:.2f} [{item.chunk.citation}]")
print(f"  {mode} (this is what the generator receives):")
for rank, item in enumerate(results, start=1):
    dense_text = "-" if item.dense is None else f"{item.dense:.2f}"
    print(
        f"    {rank}. {item.chunk.id} score={item.score:.2f} "
        f"(lex={item.lexical:.2f}, dense={dense_text}) [{item.chunk.citation}]"
    )

context = format_context(results)
user_message = f"Question: {QUESTION}\n\nExcerpts:\n{context}"
print("\n== step 7: exact context block in the user message ==")
print(f"  system prompt: {len(SYSTEM_WITH_CONTEXT)} chars | user message: {len(user_message)} chars")
print("  --- begin context ---")
print(context)
print("  --- end context ---")
print("  No LLM call here. llm.chat() would POST these two messages next.")

What already exists before you ask

ingest runs once. It cleans the PDF into labelled pages (pages.json), slices those pages into 1,404 overlapping chunks (chunks.jsonl), and builds two indexes over the chunk texts: a sparse TF-IDF matrix (lexical.joblib) and, optionally, 384-dimensional semantic vectors (dense-fastembed-BAAI_bge-small-en-v1.5.npy). A question never touches the PDF; it reads these four files.

Seven stages from question to context

  1. Settings. config.py merges environment variables, a JSON file and dataclass defaults. One detail matters here: the saved default says embed=tfidf, while the dense file on disk was built with fastembed. Both demos detect the backend from the filename so the trace actually shows hybrid retrieval.
  2. Chunks. The Retriever loads all 1,404 chunks. Every chunk carries its own page span and chapter, which is where citations come from later.
  3. Lexical scores. The question is vectorised with the saved TF-IDF vectorizer and dotted against the sparse matrix. Both vectors are length-normalised, so the result is a cosine similarity per chunk. Only 118 chunks score above zero for the example question.
  4. Dense scores. The question is embedded with bge-small-en-v1.5 and dotted against the 1,404 × 384 matrix. Again cosine similarity, but in a 384-dimensional semantic space where related wording lands close together.
  5. Fusion. The two score ranges are very different — roughly 0.00–0.17 lexical against 0.33–0.80 dense. Adding them raw would let dense dominate, so _minmax() rescales each list to 0–1 separately before the 0.50 / 0.50 weighted sum. The fused numbers are relative ranks, not probabilities.
  6. Ranking. An optional chapter filter narrows the candidate indices, then np.argsort(-final)[:k] selects the top four.
  7. Context. format_context() turns those chunks into a numbered block of excerpts with their citation headers. That block becomes the user message. Nothing later can fetch more evidence.

From context to answer

prepare() wraps the excerpts in a system prompt that says: use only these excerpts, cite every sentence, say “I don't know based on the book” when unsupported, stay under 150 words. llm.chat() hashes the backend, model, full message list, temperature, token cap and thinking flag into a cache key. On a hit it returns the stored answer; on a miss it POSTs to Ollama or an OpenAI-compatible endpoint, caches the result and logs usage. finalize() packages the text, the four passages and the response metadata into an Answer object.

"""Follow one question all the way to an Answer object, LLM call included.

Step 1: load Settings and detect the dense backend, as in demo 01.
Step 2: prepare() -> messages + passages; the retrieval-to-generation hand-off.
Step 3: compute the response-cache key and report hit or miss.
Step 4: llm.chat() -> the real HTTP call to Ollama (or an OpenAI-compatible API).
Step 5: finalize() -> Answer; print its fields and sources.
Step 6: show the side effects: cache file and the usage log line.
Step 7: call llm.chat() again to demonstrate the free cache hit.

Run from the repo root:
    .venv/bin/python demos/02_full_flow.py
    .venv/bin/python demos/02_full_flow.py "What did Asita see?"

The .txt beside this file is a snapshot: answer text depends on the model
and on cache state, so a rerun after a cache miss will differ. The demo caps
max_tokens at 300 to keep the CPU-only run short; that also changes the cache
key, so this entry is separate from a normal `ask`. Numbers print at 2 decimals.
"""

import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent.parent))

from rag import llm
from rag.config import Settings
from rag.generate import finalize, prepare
from rag.retrieve import Retriever

QUESTION = sys.argv[1] if len(sys.argv) > 1 else "Where was the bodhisatta born?"

settings = Settings.load()
dense_files = sorted(settings.index_dir.glob("dense-*.npy"))
if dense_files:
    backend = dense_files[0].stem.removeprefix("dense-").split("-")[0]
    settings = settings.overrides(embed_backend=backend, max_tokens=300)

print("== step 1: configuration ==")
print(
    f"  llm={settings.llm_backend}/{settings.llm_model} | "
    f"embed={settings.embed_backend} | top_k={settings.top_k} | "
    f"max_tokens={settings.max_tokens} | think={settings.think}"
)
print(f"  question: {QUESTION}")

retriever = Retriever(settings)
messages, passages = prepare(QUESTION, retriever, settings)
print("\n== step 2: prepare() ==")
print(f"  messages: {len(messages)} -> {[m['role'] for m in messages]}")
print(
    f"  system chars: {len(messages[0]['content'])} | "
    f"user chars: {len(messages[1]['content'])}"
)
print(f"  passages: {len(passages)} -> {[p.chunk.id for p in passages]}")
print("  full excerpt block: see demo 01, step 7")

key = llm._cache_key(settings, messages)
cache_file = llm._cache_file(settings, key)
print("\n== step 3: response cache key ==")
print(f"  key: {key}")
print(f"  file: {cache_file.name} | exists={cache_file.exists()}")

print("\n== step 4: llm.chat() ==")
response = llm.chat(settings, messages)
assert isinstance(response, llm.LLMResponse)  # stream=False returns LLMResponse, not deltas
usage = response.usage
print(
    f"  backend={response.backend} | model={response.model} | "
    f"cached={response.cached}"
)
print(
    f"  prompt_tokens={usage.get('prompt_tokens', 0)} | "
    f"completion_tokens={usage.get('completion_tokens', 0)} | "
    f"total_duration_ms={usage.get('total_duration_ms', 0)}"
)
print("  --- answer text ---")
print(response.text)
print("  --- end answer text ---")

result = finalize(QUESTION, response.text, passages, response)
print("\n== step 5: finalize() -> Answer ==")
print(f"  Answer.question: {result.question}")
print(f"  Answer.text: {len(result.text)} chars")
print(f"  Answer.passages: {len(result.passages)}")
print("  sources: " + " ".join(f"[{p.chunk.id}, {p.chunk.citation}]" for p in result.passages))

print("\n== step 6: side effects ==")
print(f"  cache file now exists: {cache_file.exists()}")
usage_path = settings.cache_dir / "usage.jsonl"
last_line = usage_path.read_text(encoding="utf-8").splitlines()[-1]
print(f"  last usage line: {last_line}")
if response.cached:
    print("  cached responses skip usage logging; that line is from an earlier call")

again = llm.chat(settings, messages)
assert isinstance(again, llm.LLMResponse)
print("\n== step 7: second identical call ==")
print(f"  cached={again.cached} | same_text={again.text == response.text}")
print("  This is why repeated experiments with the same prompt are free.")

Three things worth noticing

  • Dense retrieval reorders the same chunks: c0023 is fourth with lexical alone but third with hybrid, because dense gives it 0.80 while lexical gives it only 0.10.
  • The per-sentence citation rule and the “I don't know” fallback are prompt instructions, not Python checks. evaluate.py is what measures whether the model follows them.
  • The cache key contains the whole prompt, so changing retrieval settings, k, the model or the token cap forces a fresh answer. Demo 02's captured output is a snapshot: a rerun after the cache is populated shows cached=True.
